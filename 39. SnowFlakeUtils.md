# SnowflakeUtils
## 1. Giới thiệu
SnowflakeUtils là một thư viện của Twitter được thiết kế để tạo và phân tích các ID duy nhất gọi là Snowflakes. Snowflakes là các số nguyên 64-bit được sử dụng để xác định duy nhất các đối tượng trong hệ thống phân tán, như tweet, người dùng, v.v. Mỗi Snowflake chứa thông tin về thời gian tạo, máy chủ tạo và một số thứ tự để đảm bảo tính duy nhất.
## 2. Cấu trúc của Snowflake
Một Snowflake bao gồm các phần sau:
- 41 bit: Timestamp (thời gian tạo) - Số mili giây kể từ một mốc thời gian cố định (epoch).
- 10 bit: Machine ID (ID máy chủ) - Xác định máy chủ hoặc quy trình tạo Snowflake.
- 12 bit: Sequence (số thứ tự) - Một số tăng dần để đảm bảo tính duy nhất trong cùng một mili giây.
## 3. Sử dụng SnowflakeUtils
### 3.1 Tạo Snowflake
Để tạo một Snowflake mới, bạn có thể thực hiện như sau:
```java
package com.didan.testperformance.first.util;

import java.net.NetworkInterface;
import java.security.SecureRandom;
import java.util.Enumeration;
import org.springframework.stereotype.Component;

/**
 * Distributed Sequence Generator. Inspired by <a href="https://github.com/twitter/snowflake/tree/snowflake-2010">Twitter snowflake</a>
 * <p>
 * This class should be used as a Singleton. Make sure that you create and reuse a Single instance of SequenceGenerator per node in your distributed system cluster.
 */
@Component
public class SnowFlakeUtils {

  private static final int NODE_ID_BITS = 10; // Số bit dành cho Node ID (10 bit = 1024 node)
  private static final int SEQUENCE_BITS = 12; // Số bit dành cho Sequence (12 bit = 4096 IDs per millisecond)
  private static final long MAX_NODE_ID = (1 << NODE_ID_BITS) - 1; // Giá trị lớn nhất của Node ID = 1024 - 1
  private static final long MAX_SEQUENCE = (1 << SEQUENCE_BITS) - 1; // Giá trị lớn nhất của Sequence = 4096 - 1
  private static final long CUSTOM_EPOCH = 1609459200000L; // Giá trị epoch tùy chỉnh (là giá trị tinh bằng milliseconds từ 2021-01-01)
  private final long nodeId; // Node ID của máy chủ hiện tại
  private volatile long lastTimestamp = -1L; // Thời gian của ID cuối cùng được tạo bằng milliseconds từ epoch (với volatile để đảm bảo tính nhất quán trong môi trường đa luồng)
  private volatile long sequence = 0L; // Sequence của ID cuối cùng được tạo (với volatile để đảm bảo tính nhất quán trong môi trường đa luồng)

  public SnowFlakeUtils() {
    this.nodeId = createNodeId(); // Tạo Node ID ngẫu nhiên trong phạm vi hợp lệ
  }

  /**
   * Hàm tạo Node ID ngẫu nhiên trong phạm vi hợp lệ (synchronized đảm bảo luồng không bị xung đột khi tạo Node ID, phải chờ từng luồng hoàn thành)
   *
   * @return
   */
  public synchronized long nextId() {
    long currentTimestamp = timestamp(); // Lấy thời gian hiện tại bằng milliseconds từ epoch
    if (currentTimestamp < lastTimestamp) { // Kiểm tra nếu thời gian hiện tại với thời gian của đồng hồ trên thiết bị đang chạy chậm hơn thời gian của ID cuối cùng được tạo
      throw new IllegalStateException("Clock moved backwards. Refusing to generate id for " + (lastTimestamp - currentTimestamp) + " milliseconds");
    }
    if (lastTimestamp == currentTimestamp) { // Nếu thời gian hiện tại bằng với thời gian của ID cuối cùng được tạo
      sequence = (sequence + 1) & MAX_SEQUENCE; // Tăng sequence lên 1 và đảm bảo nó không vượt quá giá trị lớn nhất
      if (sequence == 0) { // Nếu sequence vượt quá giá trị lớn nhất, chờ đến millisecond tiếp theo
        currentTimestamp = waitNextMillis(currentTimestamp);
      }
    } else {
      sequence = 0; // Reset sequence nếu thời gian hiện tại khác với thời gian của ID cuối cùng được tạo
    }
    lastTimestamp = currentTimestamp; // Cập nhật thời gian của ID cuối cùng được tạo
    // Tạo ID bằng cách kết hợp các thành phần lại với nhau
    return currentTimestamp << (NODE_ID_BITS + SEQUENCE_BITS) // Dịch chuyển thời gian sang bên trái để dành chỗ cho Node ID và Sequence
        | (nodeId << SEQUENCE_BITS) // Dịch chuyển Node ID sang bên trái để dành chỗ cho Sequence
        | sequence; // Thêm Sequence vào cuối cùng
  }

  /**
   * Hàm createNodeId ngẫu nhiên trong phạm vi hợp lệ
   *
   * @return
   */
  private long createNodeId() {
    try {
      StringBuilder sb = new StringBuilder();
      Enumeration<NetworkInterface> networkInterfaces = NetworkInterface.getNetworkInterfaces(); // Lấy danh sách các giao diện mạng trên máy
      while (networkInterfaces.hasMoreElements()) { // Duyệt qua từng giao diện mạng
        NetworkInterface networkInterface = networkInterfaces.nextElement(); // Lấy giao diện mạng hiện tại
        byte[] mac = networkInterface.getHardwareAddress(); // Lấy địa chỉ MAC của giao diện mạng
        if (mac != null) { // Nếu địa chỉ MAC không null
          for (byte b : mac) {
            sb.append(String.format("%02X", b)); // Chuyển đổi từng byte của địa chỉ MAC thành chuỗi hex và thêm vào StringBuilder
          }
        }
      }
      return sb.toString().hashCode() & MAX_NODE_ID; // Trả về mã băm của chuỗi địa chỉ MAC và đảm bảo nó nằm trong phạm vi hợp lệ
    } catch (Exception ex) {
      return (new SecureRandom().nextLong()) & MAX_NODE_ID; // Nếu có lỗi xảy ra, trả về một số ngẫu nhiên trong phạm vi hợp lệ
    }
  }

  /**
   * Hàm tạo thời gian hiện tại bằng milliseconds từ epoch tùy chỉnh
   * @return
   */
  private long timestamp() {
    return System.currentTimeMillis() - CUSTOM_EPOCH; // Trả về thời gian hiện tại bằng milliseconds từ epoch tùy chỉnh
  }

  /**
   * Hàm chờ đến millisecond tiếp theo
   * @param currentTimestamp
   * @return
   */
  private long waitNextMillis(long currentTimestamp) {
    while (currentTimestamp == lastTimestamp) { // Chờ đến khi thời gian hiện tại khác với thời gian của ID cuối cùng được tạo
      currentTimestamp = timestamp(); // Cập nhật thời gian hiện tại
    }
    return currentTimestamp; // Trả về thời gian hiện tại
  }
}
```
